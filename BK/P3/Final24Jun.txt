function createCATK(dx, dy, dr, ctx)
{
    ctx.beginPath();  //tell canvas to start a set of lines
        
        point(dx,dy,ctx);             //CENTER

    ctx.stroke();          //actually draw the accumulated lines
}
  
  
function createD(dx, dy, dr, ctx)
{
    var step = 2*Math.PI / 180;
    ctx.beginPath();  //tell canvas to start a set of lines
    for(var dtheta = 0*Math.PI/180;  dtheta < 91*Math.PI/180;  dtheta+=step)  { 
	  var x = dx + dr*Math.cos(dtheta);
      var y = dy - dr*Math.sin(dtheta);    //note 2.
      ctx.lineWidth=2;
      ctx.lineTo(x,y);
    }
    ctx.stroke();          //actually draw the accumulated lines
      
    ctx.beginPath();  //tell canvas to start a set of lines
    for(var dtheta = 270*Math.PI/180;  dtheta < 360*Math.PI/180;  dtheta+=step)  { 
	  var x = dx + dr*Math.cos(dtheta);
      var y = dy - dr*Math.sin(dtheta);    //note 2.
      ctx.lineWidth=2;
      ctx.lineTo(x,y);
    }
    ctx.stroke();          //actually draw the accumulated lines
        
    line(dx-(dr/3),dy-dr,dx-(dr/3),dy+dr, ctx);
    line(dx-(dr/3),dy-dr,dx,dy-dr, ctx);
    line(dx-(dr/3),dy+dr,dx,dy+dr, ctx);
}

function point(x, y, ctx){
  ctx.beginPath();
  ctx.arc(x, y, 2, 0, 2 * Math.PI, true);
  ctx.fill();
}
    
    
function line(x, y, x1, y1, canvas){
  canvas.beginPath();
  canvas.moveTo(x, y);
  canvas.lineWidth=2;
  canvas.lineTo(x1, y1);
  canvas.stroke();
}
    
function rect(x, y, h,w, canvas){
